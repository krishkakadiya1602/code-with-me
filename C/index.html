1. C Introduction

C is a general-purpose programming language used to develop software, operating systems, embedded systems and many other applications.

Important: C is a procedural programming language and is widely used for learning programming fundamentals.

First C Program

#include <stdio.h>

int main()
{
    printf("Hello World!");

    return 0;
}

2. C Installation

To run C programs, you need a C compiler. Popular C compilers include GCC and Clang.

Example

gcc program.c -o program

./program

3. C Syntax

C programs are generally written using functions, statements and expressions. 

#include <stdio.h>

int main()

5. C Variables

Variables are used to store data in memory. 

#include <stdio.h>

int main()
{
    int age = 20;

    printf("%d", age);

    return 0;
}

7. C Constants

A constant is a value that should not change during program execution. 

#include <stdio.h>

int main()
{
    const int age = 18;

    printf("%d", age);

    return 0;
}

8. C Operators

Operators are symbols used to perform operations on values and variables. 

int a = 10;
int b = 5;

printf("%d", a + b);
printf("%d", a - b);
printf("%d", a * b);
printf("%d", a / b);

9. C If Else

The if-else statement is used for decision making. 

#include <stdio.h>

int main()
{
    int age = 20;

    if(age >= 18)
    {
        printf("Adult");
    }
    else
    {
        printf("Minor");
    }

    return 0;
}

10. C Switch

The switch statement is used to select one option from multiple cases. 

#include <stdio.h>

int main()
{
    int number = 2;

    switch(number)
    {
        case 1:
            printf("One");
            break;

        case 2:
            printf("Two");
            break;

        default:
            printf("Other");
    }

    return 0;
}

11. C Loops

Loops are used to execute a block of code repeatedly.

For Loop

#include <stdio.h>

int main()
{
    int i;

    for(i = 1; i <= 5; i++)
    {
        printf("%d\n", i);
    }

    return 0;
}

While Loop

int i = 1;

while(i <= 5)
{
    printf("%d\n", i);

    i++;
}

12. C Functions

Functions are blocks of code designed to perform a particular task. 

#include <stdio.h>

void hello()
{
    printf("Hello World!");
}

int main()
{
    hello();

    return 0;
}

13. C Arrays

An array stores multiple values of the same data type. 

#include <stdio.h>

int main()
{
    int numbers[5] = {10, 20, 30, 40, 50};

    printf("%d", numbers[0]);

    return 0;
}

14. C Strings

A string is a sequence of characters. 

#include <stdio.h>

int main()
{
    char name[] = "Code With Me";

    printf("%s", name);

    return 0;
}

15. C Pointers

A pointer is a variable that stores the memory address of another variable. 

#include <stdio.h>

int main()
{
    int number = 10;

    int *ptr = &number;

    printf("%d", *ptr);

    return 0;
}

16. C Structures

Structures allow you to group different types of data together. 

#include <stdio.h>

struct Student
{
    int age;
    char grade;
};

int main()
{
    struct Student s1;

    s1.age = 20;
    s1.grade = 'A';

    printf("%d\n", s1.age);
    printf("%c", s1.grade);

    return 0;
}
