Project 1: Simple Calculator

This project uses variables, switch statements and arithmetic operators to create a basic calculator.

Code

#include <stdio.h>

int main()
{
    char operator;
    double num1, num2;

    printf("Enter an operator (+, -, *, /): ");
    scanf(" %c", &operator);

    printf("Enter two numbers: ");
    scanf("%lf %lf", &num1, &num2);

    switch(operator)
    {
        case '+':
            printf("Result = %.2lf", num1 + num2);
            break;

        case '-':
            printf("Result = %.2lf", num1 - num2);
            break;

        case '*':
            printf("Result = %.2lf", num1 * num2);
            break;

        case '/':
            if(num2 != 0)
                printf("Result = %.2lf", num1 / num2);
            else
                printf("Cannot divide by zero");
            break;

        default:
            printf("Invalid operator");
    }

    return 0;
}

Example Output

Project 2: Number Guessing Game

In this project, the computer generates a random number and the user tries to guess it.

Code

#include <stdio.h>
#include <stdlib.h>
#include <time.h>

int main()
{
    int secretNumber;
    int guess;

    srand(time(NULL));

    secretNumber = rand() % 100 + 1;

    printf("Guess a number between 1 and 100\n");

    do
    {
        printf("Enter your guess: ");
        scanf("%d", &guess);

        if(guess < secretNumber)
        {
            printf("Too low!\n");
        }
        else if(guess > secretNumber)
        {
            printf("Too high!\n");
        }
        else
        {
            printf("Correct! You guessed the number.\n");
        }

    } while(guess != secretNumber);

    return 0;
}

Concepts Used

  • Variables
  • if-else
  • do-while loop
  • rand()
  • srand()
  • Input and output

Project 3: Student Grade Calculator

This program accepts marks for three subjects and calculates the total, percentage and grade.

Code

#include <stdio.h>

int main()
{
    float maths;
    float science;
    float english;

    float total;
    float percentage;

    printf("Enter Maths marks: ");
    scanf("%f", &maths);

    printf("Enter Science marks: ");
    scanf("%f", &science);

    printf("Enter English marks: ");
    scanf("%f", &english);

    total = maths + science + english;

    percentage = total / 3;

    printf("\nTotal = %.2f\n", total);
    printf("Percentage = %.2f%%\n", percentage);

    if(percentage >= 90)
        printf("Grade = A");
    else if(percentage >= 75)
        printf("Grade = B");
    else if(percentage >= 60)
        printf("Grade = C");
    else if(percentage >= 40)
        printf("Grade = D");
    else
        printf("Grade = F");

    return 0;
}
Enter Maths marks: 85 Enter Science marks: 78 Enter English marks: 90 Total = 253.00 Percentage = 84.33% Grade = B

Project 4: Simple ATM

This project demonstrates a menu-driven ATM program using switch statements and loops.

Code

#include <stdio.h>

int main()
{
    int choice;
    float balance = 5000;
    float amount;

    do
    {
        printf("\n\n===== ATM MENU =====\n");
        printf("1. Check Balance\n");
        printf("2. Deposit\n");
        printf("3. Withdraw\n");
        printf("4. Exit\n");

        printf("Enter your choice: ");
        scanf("%d", &choice);

        switch(choice)
        {
            case 1:
                printf("Balance = %.2f", balance);
                break;

            case 2:
                printf("Enter deposit amount: ");
                scanf("%f", &amount);

                if(amount > 0)
                {
                    balance += amount;
                    printf("Deposit successful.");
                }
                else
                {
                    printf("Invalid amount.");
                }

                break;

            case 3:
                printf("Enter withdrawal amount: ");
                scanf("%f", &amount);

                if(amount > 0 && amount <= balance)
                {
                    balance -= amount;
                    printf("Withdrawal successful.");
                }
                else
                {
                    printf("Insufficient balance or invalid amount.");
                }

                break;

            case 4:
                printf("Thank you for using the ATM.");
                break;

            default:
                printf("Invalid choice.");
        }

    } while(choice != 4);

    return 0;
}

Project 5: Student Management System

This project introduces the use of structures, arrays and functions to store student information.

Code

#include <stdio.h>

struct Student
{
    int rollNo;
    char name[50];
    float marks;
};

int main()
{
    struct Student students[3];

    for(int i = 0; i < 3; i++)
    {
        printf("\nEnter student %d details:\n", i + 1);

        printf("Roll Number: ");
        scanf("%d", &students[i].rollNo);

        printf("Name: ");
        scanf("%49s", students[i].name);

        printf("Marks: ");
        scanf("%f", &students[i].marks);
    }

    printf("\n===== Student Records =====\n");

    for(int i = 0; i < 3; i++)
    {
        printf("\nRoll Number: %d", students[i].rollNo);
        printf("\nName: %s", students[i].name);
        printf("\nMarks: %.2f\n", students[i].marks);
    }

    return 0;
}

Concepts Used

  • Structures
  • Arrays
  • Strings

Project 6: Contact Management System

Project 7: Number Analysis Program

This project uses arrays to find the sum, average, largest and smallest values. 

#include <stdio.h>

int main()
{
    int numbers[5];
    int sum = 0;
    int largest;
    int smallest;

    printf("Enter 5 numbers:\n");

    for(int i = 0; i < 5; i++)
    {
        scanf("%d", &numbers[i]);
    }

    largest = numbers[0];
    smallest = numbers[0];

    for(int i = 0; i < 5; i++)
    {
        sum += numbers[i];

        if(numbers[i] > largest)
            largest = numbers[i];

        if(numbers[i] < smallest)
            smallest = numbers[i];
    }

    printf("\nSum = %d", sum);
    printf("\nAverage = %.2f", (float)sum / 5);
    printf("\nLargest = %d", largest);
    printf("\nSmallest = %d", smallest);

    return 0;
}
Enter 5 numbers: 10 50 20 40 30 Sum = 150 Average = 30.00 Largest = 50 Smallest = 10

How to Improve Your C Projects

  1. Add functions instead of putting everything inside main().
  2. Use structures when working with related data.
  3. Use files to permanently store data.
  4. Add input validation.
  5. Create menus for multiple operations.
  6. Use dynamic memory when appropriate.
  7. Split large programs into multiple functions.
  8. Test the program with different inputs.
